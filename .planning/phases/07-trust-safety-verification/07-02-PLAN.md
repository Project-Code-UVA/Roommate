import { createClient } from "@supabase/supabase-js";

const SUPABASE_URL = process.env.SUPABASE_URL ?? process.env.EXPO_PUBLIC_SUPABASE_URL ?? "";
const SUPABASE_SERVICE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY ?? "";

**createTestUser helper:**
- Accepts `overrides?: { email?: string; enforcement_state?: string; selfie_verified?: boolean; onboarding_completed?: boolean; display_name?: string }`
- Generates a unique email: `test-${Date.now()}-${Math.random().toString(36).slice(2)}@test.room.app`
- Creates auth user via `supabaseAdmin.auth.admin.createUser({ email, password: "test-password-123!", email_confirm: true })`
- Inserts into `public.users`: `{ id: authUser.id, birthdate: "2000-01-01", enforcement_state: overrides?.enforcement_state ?? "none", selfie_verified: overrides?.selfie_verified ?? false, onboarding_completed: overrides?.onboarding_completed ?? true }`
- Inserts into `public.profiles`: `{ user_id: authUser.id, display_name: overrides?.display_name ?? "Test User", year: "2026", bio: "Test bio" }`
- Returns `{ id: authUser.id, email }`

**addUserToSchool helper:**
- Takes `userId: string, schoolId?: string`
- Defaults `schoolId` to a known seeded school ID. To get a school ID: `const { data } = await supabaseAdmin.from("schools").select("id").limit(1).single()` and cache it.
- Inserts into `user_schools`: `{ user_id: userId, school_id: schoolId }`

**createTestThread helper:**
- Takes `userAId: string, userBId: string`
- Creates a match: `INSERT INTO matches (user_a_id, user_b_id) VALUES (LEAST(a,b), GREATEST(a,b))` returning match id
- Creates a thread: `INSERT INTO threads (match_id, user_a_id, user_b_id, status) VALUES (matchId, LEAST(a,b), GREATEST(a,b), 'active')` returning thread id
- Returns `{ matchId, threadId }`

**cleanupTestUsers helper:**
- Takes `userIds: string[]`
- For each user: `supabaseAdmin.auth.admin.deleteUser(userId)` -- cascading FK constraints clean up all related tables (profiles, user_schools, likes, matches, etc.)

**setEnforcementState helper:**
- Takes `userId: string, state: string`
- `supabaseAdmin.from("users").update({ enforcement_state: state }).eq("id", userId)`

**getSchoolIds helper (cached):**
- Fetches 2 distinct school IDs from the seeded schools table
- Caches result in module-level variable so it only fetches once
- Returns `{ schoolA: string, schoolB: string }` -- two different schools for testing shared vs non-shared school scenarios

All helpers use `supabaseAdmin` (service role) to bypass RLS for test setup/teardown.
